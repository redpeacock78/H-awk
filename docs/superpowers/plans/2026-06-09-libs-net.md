# libs/net Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace gawk's `/inet/tcp/` with a Zig-backed async TCP server (`libs/net`) that handles multiple simultaneous HTTP/1.1 connections, with graceful fallback when the extension is absent.

**Architecture:** Zig spawns a background event loop thread (kqueue on macOS, epoll on Linux) that accepts N simultaneous TCP connections, parses HTTP/1.1 requests, and queues them. AWK calls `hawk_net_poll()` (blocks until a request is ready) and `hawk_net_respond()` (queues a response back to the event loop for async write). The three extension functions (`hawk_net_listen`, `hawk_net_poll`, `hawk_net_respond`) are registered via `ffi.makeDlLoad` following the same pattern as `libs/binary`.

**Tech Stack:** Zig 0.14 (same as rest of project), gawk extension API (`gawk_ffi.zig`), kqueue (macOS) / epoll (Linux) via `std.posix`, AWK (`core/http.awk`, `core/request.awk`)

## Poll-Result Wire Format

`hawk_net_poll()` returns a string with fields separated by ASCII `\x1e` (Record Separator, 0x1E):

```
conn_id RS method RS path RS headers_block RS body_len RS body
```

Example: `42\x1eGET\x1e/hello\x1eHost: localhost\r\n\x1e0\x1e`

AWK splits on `\x1e`. Body is reconstructed by joining `parts[6..n]` with `\x1e` to handle the rare case where body itself contains `\x1e`.

```zig
// libs/net/src/http_parser.zig
// SPDX-License-Identifier: MIT
const std = @import("std");

pub const Request = struct {
    conn_id: u64,
    method: []const u8,
    path: []const u8,
    headers_block: []const u8,
    body: []const u8,
    keep_alive: bool,
    content_length: usize,

    pub fn deinit(self: Request, alloc: std.mem.Allocator) void {
        alloc.free(self.method);
        alloc.free(self.path);
        alloc.free(self.headers_block);
        alloc.free(self.body);
    }
};

/// Parse raw HTTP/1.1 request bytes. Returns owned Request; caller calls deinit.
pub fn parse(buf: []const u8, conn_id: u64, alloc: std.mem.Allocator) !Request {
    const sep = std.mem.indexOf(u8, buf, "\r\n\r\n") orelse return error.InvalidRequest;
    const headers_section = buf[0..sep];
    const body_raw = buf[sep + 4 ..];

    const first_crlf = std.mem.indexOf(u8, headers_section, "\r\n") orelse headers_section.len;
    const request_line = headers_section[0..first_crlf];

    var it = std.mem.splitScalar(u8, request_line, ' ');
    const method = it.next() orelse return error.InvalidRequest;
    const path = it.next() orelse return error.InvalidRequest;
    if (method.len == 0 or path.len == 0) return error.InvalidRequest;

    const headers_block = if (first_crlf + 2 <= headers_section.len)
        headers_section[first_crlf + 2 ..]
    else
        "";

    var content_length: usize = 0;
    var keep_alive = false;
    var hdr_it = std.mem.splitSequence(u8, headers_block, "\r\n");
    while (hdr_it.next()) |hline| {
        if (hline.len == 0) continue;
        const colon = std.mem.indexOf(u8, hline, ":") orelse continue;
        const hname = std.mem.trim(u8, hline[0..colon], " ");
        const hval = std.mem.trim(u8, hline[colon + 1 ..], " ");
        if (std.ascii.eqlIgnoreCase(hname, "content-length")) {
            content_length = std.fmt.parseInt(usize, hval, 10) catch 0;
        }
        if (std.ascii.eqlIgnoreCase(hname, "connection")) {
            keep_alive = std.ascii.eqlIgnoreCase(hval, "keep-alive");
        }
    }

    const body = body_raw[0..@min(content_length, body_raw.len)];

    return .{
        .conn_id = conn_id,
        .method = try alloc.dupe(u8, method),
        .path = try alloc.dupe(u8, path),
        .headers_block = try alloc.dupe(u8, headers_block),
        .body = try alloc.dupe(u8, body),
        .keep_alive = keep_alive,
        .content_length = content_length,
    };
}

/// Format poll result string for AWK.
/// Format: conn_id RS method RS path RS headers_block RS body_len RS body
/// RS = \x1e (ASCII Record Separator). Caller owns returned slice.
pub fn formatPollResult(req: Request, alloc: std.mem.Allocator) ![]const u8 {
    return std.fmt.allocPrint(alloc, "{d}\x1e{s}\x1e{s}\x1e{s}\x1e{d}\x1e{s}", .{
        req.conn_id,
        req.method,
        req.path,
        req.headers_block,
        req.body.len,
        req.body,
    });
}
```

```zig
// libs/net/src/conn_pool.zig
// SPDX-License-Identifier: MIT
const std = @import("std");

pub const ConnState = enum { reading, ready, closed };

pub const Conn = struct {
    fd: std.posix.socket_t,
    state: ConnState,
    read_buf: std.ArrayList(u8),
    keep_alive: bool,
};

pub const ConnPool = struct {
    alloc: std.mem.Allocator,
    conns: std.AutoHashMap(u64, Conn),
    next_id: u64,
    mu: std.Thread.Mutex,

    pub fn init(alloc: std.mem.Allocator) ConnPool {
        return .{
            .alloc = alloc,
            .conns = std.AutoHashMap(u64, Conn).init(alloc),
            .next_id = 1,
            .mu = .{},
        };
    }

    pub fn deinit(self: *ConnPool) void {
        var it = self.conns.valueIterator();
        while (it.next()) |conn| {
            conn.read_buf.deinit();
            std.posix.close(conn.fd);
        }
        self.conns.deinit();
    }

    /// Add a new connection. Returns conn_id. Caller owns fd lifecycle via pool.
    pub fn add(self: *ConnPool, fd: std.posix.socket_t) !u64 {
        self.mu.lock();
        defer self.mu.unlock();
        const id = self.next_id;
        self.next_id += 1;
        try self.conns.put(id, .{
            .fd = fd,
            .state = .reading,
            .read_buf = std.ArrayList(u8).init(self.alloc),
            .keep_alive = false,
        });
        return id;
    }

    /// Remove and close a connection. No-op if id not found.
    pub fn remove(self: *ConnPool, id: u64) void {
        self.mu.lock();
        defer self.mu.unlock();
        if (self.conns.fetchRemove(id)) |kv| {
            var conn = kv.value;
            conn.read_buf.deinit();
            std.posix.close(conn.fd);
        }
    }

    pub fn count(self: *ConnPool) usize {
        self.mu.lock();
        defer self.mu.unlock();
        return self.conns.count();
    }
};
```

The event loop has two threads:
- **Event loop thread** (Zig): accepts connections, reads data, parses HTTP, enqueues poll strings
- **AWK thread** (main): calls `dequeue()` (blocks), then calls `respond()` (queues response)

Response path: `respond()` puts `PendingResponse` into `resp_queue`, then writes one byte to `wakeup_write` pipe to interrupt the kqueue/epoll wait. The event loop drains `resp_queue` on wakeup.

```zig
// libs/net/src/event_loop.zig
// SPDX-License-Identifier: MIT
const std = @import("std");
const builtin = @import("builtin");
const http_parser = @import("http_parser");
const conn_pool = @import("conn_pool");

const BACKLOG: u31 = 128;
const MAX_EVENTS = 64;
const READ_BUF_SIZE = 65536;

pub const PendingResponse = struct {
    conn_id: u64,
    data: []u8,
};

pub const EventLoop = struct {
    alloc: std.mem.Allocator,
    listen_fd: std.posix.socket_t,
    pool: conn_pool.ConnPool,

    req_queue: std.ArrayList([]u8),
    req_mu: std.Thread.Mutex,
    req_cond: std.Thread.Condition,

    resp_queue: std.ArrayList(PendingResponse),
    resp_mu: std.Thread.Mutex,

    // Pipe used to wake up the event loop when responses are queued.
    // Event loop monitors wakeup_read_fd; respond() writes one byte to wakeup_write_fd.
    wakeup_read_fd: std.posix.fd_t,
    wakeup_write_fd: std.posix.fd_t,

    running: std.atomic.Value(bool),

    pub fn init(alloc: std.mem.Allocator, port: u16) !EventLoop {
        const addr = std.net.Address.initIp4(.{ 0, 0, 0, 0 }, port);
        const listen_fd = try std.posix.socket(
            std.posix.AF.INET,
            std.posix.SOCK.STREAM | std.posix.SOCK.CLOEXEC,
            0,
        );
        errdefer std.posix.close(listen_fd);
        try std.posix.setsockopt(listen_fd, std.posix.SOL.SOCKET, std.posix.SO.REUSEADDR, &std.mem.toBytes(@as(c_int, 1)));
        var sock_addr = addr.any;
        try std.posix.bind(listen_fd, &sock_addr, addr.getOsSockLen());
        try std.posix.listen(listen_fd, BACKLOG);

        const pipe_fds = try std.posix.pipe2(.{ .CLOEXEC = true });

        return .{
            .alloc = alloc,
            .listen_fd = listen_fd,
            .pool = conn_pool.ConnPool.init(alloc),
            .req_queue = std.ArrayList([]u8).init(alloc),
            .req_mu = .{},
            .req_cond = .{},
            .resp_queue = std.ArrayList(PendingResponse).init(alloc),
            .resp_mu = .{},
            .wakeup_read_fd = pipe_fds[0],
            .wakeup_write_fd = pipe_fds[1],
            .running = std.atomic.Value(bool).init(true),
        };
    }

    pub fn deinit(self: *EventLoop) void {
        std.posix.close(self.listen_fd);
        std.posix.close(self.wakeup_read_fd);
        std.posix.close(self.wakeup_write_fd);
        self.pool.deinit();
        for (self.req_queue.items) |item| self.alloc.free(item);
        self.req_queue.deinit();
        for (self.resp_queue.items) |item| self.alloc.free(item.data);
        self.resp_queue.deinit();
    }

    pub fn stop(self: *EventLoop) void {
        self.running.store(false, .release);
        // Wake up event loop so it notices running=false
        _ = std.posix.write(self.wakeup_write_fd, "x") catch {};
        // Wake up any blocked dequeue() call
        self.req_mu.lock();
        self.req_cond.signal();
        self.req_mu.unlock();
    }

    /// Block until a request is available. Returns poll string owned by gawk_alloc.
    /// Returns null if event loop stopped.
    pub fn dequeue(self: *EventLoop, gawk_alloc: std.mem.Allocator) ?[]u8 {
        self.req_mu.lock();
        defer self.req_mu.unlock();
        while (self.req_queue.items.len == 0) {
            if (!self.running.load(.acquire)) return null;
            self.req_cond.wait(&self.req_mu);
        }
        const item = self.req_queue.orderedRemove(0);
        const result = gawk_alloc.dupe(u8, item) catch return null;
        self.alloc.free(item);
        return result;
    }

    /// Queue a response for async write by the event loop thread.
    /// data is the full HTTP response wire (status + headers + body).
    pub fn respond(self: *EventLoop, conn_id: u64, data: []const u8) bool {
        const owned = self.alloc.dupe(u8, data) catch return false;
        {
            self.resp_mu.lock();
            defer self.resp_mu.unlock();
            self.resp_queue.append(.{ .conn_id = conn_id, .data = owned }) catch {
                self.alloc.free(owned);
                return false;
            };
        }
        _ = std.posix.write(self.wakeup_write_fd, "x") catch {};
        return true;
    }

    /// Run the event loop. Called in background thread.
    pub fn run(self: *EventLoop) void {
        if (builtin.os.tag == .macos or builtin.os.tag == .ios) {
            self.runKqueue() catch |err| {
                std.log.err("kqueue event loop failed: {}", .{err});
            };
        } else {
            self.runEpoll() catch |err| {
                std.log.err("epoll event loop failed: {}", .{err});
            };
        }
        // Signal blocked dequeue() callers
        self.req_mu.lock();
        self.req_cond.broadcast();
        self.req_mu.unlock();
    }

    fn enqueueRequest(self: *EventLoop, poll_str: []const u8) !void {
        const owned = try self.alloc.dupe(u8, poll_str);
        self.req_mu.lock();
        defer self.req_mu.unlock();
        try self.req_queue.append(owned);
        self.req_cond.signal();
    }

    fn drainResponses(self: *EventLoop) void {
        self.resp_mu.lock();
        if (self.resp_queue.items.len == 0) {
            self.resp_mu.unlock();
            return;
        }
        const items = self.resp_queue.toOwnedSlice() catch {
            self.resp_mu.unlock();
            return;
        };
        self.resp_mu.unlock();
        defer self.alloc.free(items);

        for (items) |pending| {
            defer self.alloc.free(pending.data);
            self.pool.mu.lock();
            const conn = self.pool.conns.getPtr(pending.conn_id);
            const fd = if (conn) |c| c.fd else null;
            const keep_alive = if (conn) |c| c.keep_alive else false;
            self.pool.mu.unlock();

            if (fd) |sock_fd| {
                _ = std.posix.write(sock_fd, pending.data) catch {};
                if (!keep_alive) {
                    self.pool.remove(pending.conn_id);
                }
            }
        }
    }

    fn handleRead(self: *EventLoop, conn_id: u64) void {
        self.pool.mu.lock();
        const conn_ptr = self.pool.conns.getPtr(conn_id);
        const fd = if (conn_ptr) |c| c.fd else null;
        self.pool.mu.unlock();
        const sock_fd = fd orelse return;

        var tmp: [READ_BUF_SIZE]u8 = undefined;
        const n = std.posix.read(sock_fd, &tmp) catch {
            self.pool.remove(conn_id);
            return;
        };
        if (n == 0) {
            self.pool.remove(conn_id);
            return;
        }

        // Append bytes and check for complete request
        self.pool.mu.lock();
        const conn = self.pool.conns.getPtr(conn_id) orelse {
            self.pool.mu.unlock();
            return;
        };
        conn.read_buf.appendSlice(tmp[0..n]) catch {
            self.pool.mu.unlock();
            self.pool.remove(conn_id);
            return;
        };

        // Check for \r\n\r\n
        const header_end = std.mem.indexOf(u8, conn.read_buf.items, "\r\n\r\n");
        if (header_end == null) {
            self.pool.mu.unlock();
            return;
        }

        // Parse content-length
        const headers_section = conn.read_buf.items[0..header_end.?];
        var content_length: usize = 0;
        var hdr_it = std.mem.splitSequence(u8, headers_section, "\r\n");
        _ = hdr_it.next(); // skip request line
        while (hdr_it.next()) |hline| {
            const colon = std.mem.indexOf(u8, hline, ":") orelse continue;
            const hname = std.mem.trim(u8, hline[0..colon], " ");
            if (std.ascii.eqlIgnoreCase(hname, "content-length")) {
                const hval = std.mem.trim(u8, hline[colon + 1 ..], " ");
                content_length = std.fmt.parseInt(usize, hval, 10) catch 0;
            }
        }

        const total_needed = header_end.? + 4 + content_length;
        if (conn.read_buf.items.len < total_needed) {
            self.pool.mu.unlock();
            return;
        }

        const raw = conn.read_buf.items[0..total_needed];
        const req = http_parser.parse(raw, conn_id, self.alloc) catch {
            self.pool.mu.unlock();
            self.pool.remove(conn_id);
            return;
        };
        const remaining_start = total_needed;
        const remaining = conn.read_buf.items[remaining_start..];
        var new_buf = std.ArrayList(u8).init(self.alloc);
        new_buf.appendSlice(remaining) catch {};
        conn.read_buf.deinit();
        conn.read_buf = new_buf;
        conn.keep_alive = req.keep_alive;
        self.pool.mu.unlock();

        defer req.deinit(self.alloc);
        const poll_str = http_parser.formatPollResult(req, self.alloc) catch return;
        defer self.alloc.free(poll_str);
        self.enqueueRequest(poll_str) catch {};
    }

    fn runKqueue(self: *EventLoop) !void {
        const kq = try std.posix.kqueue();
        defer std.posix.close(kq);

        // Register listen_fd and wakeup_read_fd
        const initial_evs = [_]std.posix.Kevent{
            .{
                .ident = @intCast(self.listen_fd),
                .filter = std.posix.EVFILT_READ,
                .flags = std.posix.EV_ADD | std.posix.EV_ENABLE,
                .fflags = 0,
                .data = 0,
                .udata = 0,  // 0 = listen_fd sentinel
            },
            .{
                .ident = @intCast(self.wakeup_read_fd),
                .filter = std.posix.EVFILT_READ,
                .flags = std.posix.EV_ADD | std.posix.EV_ENABLE,
                .fflags = 0,
                .data = 0,
                .udata = std.math.maxInt(usize),  // maxInt = wakeup sentinel
            },
        };
        _ = try std.posix.kevent(kq, &initial_evs, &.{}, null);

        var events: [MAX_EVENTS]std.posix.Kevent = undefined;

        while (self.running.load(.acquire)) {
            const ts = std.posix.timespec{ .tv_sec = 1, .tv_nsec = 0 };
            const n = std.posix.kevent(kq, &.{}, &events, &ts) catch continue;

            self.drainResponses();

            for (events[0..n]) |ev| {
                if (ev.udata == std.math.maxInt(usize)) {
                    // Wakeup pipe - drain the byte
                    var drain: [64]u8 = undefined;
                    _ = std.posix.read(self.wakeup_read_fd, &drain) catch {};
                    continue;
                }
                if (ev.udata == 0) {
                    // New connection on listen_fd
                    const client_fd = std.posix.accept(self.listen_fd, null, null) catch continue;
                    const conn_id = self.pool.add(client_fd) catch {
                        std.posix.close(client_fd);
                        continue;
                    };
                    const client_ev = std.posix.Kevent{
                        .ident = @intCast(client_fd),
                        .filter = std.posix.EVFILT_READ,
                        .flags = std.posix.EV_ADD | std.posix.EV_ENABLE,
                        .fflags = 0,
                        .data = 0,
                        .udata = @intCast(conn_id),
                    };
                    _ = std.posix.kevent(kq, &.{client_ev}, &.{}, null) catch {
                        self.pool.remove(conn_id);
                        continue;
                    };
                } else {
                    // Data on client connection
                    const conn_id: u64 = @intCast(ev.udata);
                    self.handleRead(conn_id);
                }
            }
        }
    }

    fn runEpoll(self: *EventLoop) !void {
        const ep = try std.posix.epoll_create1(std.posix.EPOLL_CLOEXEC);
        defer std.posix.close(ep);

        // listen_fd: use fd as data (negative sentinel won't collide with conn_ids)
        var listen_ev = std.posix.epoll_event{
            .events = std.posix.EPOLLIN,
            .data = .{ .u64 = std.math.maxInt(u64) - 1 },
        };
        try std.posix.epoll_ctl(ep, std.posix.EPOLL_CTL_ADD, self.listen_fd, &listen_ev);

        // wakeup pipe
        var wakeup_ev = std.posix.epoll_event{
            .events = std.posix.EPOLLIN,
            .data = .{ .u64 = std.math.maxInt(u64) },
        };
        try std.posix.epoll_ctl(ep, std.posix.EPOLL_CTL_ADD, self.wakeup_read_fd, &wakeup_ev);

        var events: [MAX_EVENTS]std.posix.epoll_event = undefined;

        while (self.running.load(.acquire)) {
            const n = std.posix.epoll_wait(ep, &events, 1000) catch continue;

            self.drainResponses();

            for (events[0..n]) |ev| {
                const token = ev.data.u64;
                if (token == std.math.maxInt(u64)) {
                    // Wakeup pipe
                    var drain: [64]u8 = undefined;
                    _ = std.posix.read(self.wakeup_read_fd, &drain) catch {};
                } else if (token == std.math.maxInt(u64) - 1) {
                    // New connection
                    const client_fd = std.posix.accept(self.listen_fd, null, null) catch continue;
                    const conn_id = self.pool.add(client_fd) catch {
                        std.posix.close(client_fd);
                        continue;
                    };
                    var client_ev = std.posix.epoll_event{
                        .events = std.posix.EPOLLIN | std.posix.EPOLLET,
                        .data = .{ .u64 = conn_id },
                    };
                    std.posix.epoll_ctl(ep, std.posix.EPOLL_CTL_ADD, client_fd, &client_ev) catch {
                        self.pool.remove(conn_id);
                        continue;
                    };
                } else {
                    self.handleRead(token);
                }
            }
        }
    }
};
```

```zig
// SPDX-License-Identifier: MIT
const std = @import("std");
const ffi = @import("gawk_ffi");
const event_loop = @import("event_loop");

// Global state: one EventLoop per process (gawk is single-process).
var _loop: ?*event_loop.EventLoop = null;
var _loop_thread: ?std.Thread = null;
var _loop_alloc = std.heap.c_allocator;

const _ffi_entry = ffi.makeDlLoad(.{
    .name = "hawk_net",
    .functions = &.{
        .{ .name = "hawk_net_listen",  .impl = &netListen,  .args = 1 },
        .{ .name = "hawk_net_poll",    .impl = &netPoll,    .args = 0 },
        .{ .name = "hawk_net_respond", .impl = &netRespond, .args = 4 },
    },
});
comptime {
    _ = _ffi_entry;
}

fn netListen(args: ffi.Args) ffi.Result {
    if (_loop != null) return .{ .int = 1 };  // idempotent
    const port_val = args.getInt(0);
    if (port_val <= 0 or port_val > 65535) return .{ .int = 0 };
    const port: u16 = @intCast(port_val);

    const loop = _loop_alloc.create(event_loop.EventLoop) catch return .{ .int = 0 };
    loop.* = event_loop.EventLoop.init(_loop_alloc, port) catch {
        _loop_alloc.destroy(loop);
        return .{ .int = 0 };
    };
    _loop = loop;

    _loop_thread = std.Thread.spawn(.{}, event_loop.EventLoop.run, .{loop}) catch {
        loop.deinit();
        _loop_alloc.destroy(loop);
        _loop = null;
        return .{ .int = 0 };
    };

    return .{ .int = 1 };
}

fn netPoll(_: ffi.Args) ffi.Result {
    const loop = _loop orelse return .none;
    // Blocks until a request is ready. Returns gawk-owned string (via gawkAllocator).
    const result = loop.dequeue(ffi.gawkAllocator()) orelse return .none;
    return .{ .string = result };
}

fn netRespond(args: ffi.Args) ffi.Result {
    const loop = _loop orelse return .{ .int = 0 };
    const conn_id_str = args.getString(0);
    const status_line = args.getString(1);
    const headers = args.getString(2);
    const body = args.getString(3);

    const conn_id = std.fmt.parseInt(u64, conn_id_str, 10) catch return .{ .int = 0 };

    // Build HTTP response wire: "status_line\r\nheaders\r\nbody"
    // headers is expected to already end with \r\n for each header line.
    const data = std.fmt.allocPrint(_loop_alloc, "{s}\r\n{s}\r\n{s}", .{
        status_line, headers, body,
    }) catch return .{ .int = 0 };
    defer _loop_alloc.free(data);

    return .{ .int = if (loop.respond(conn_id, data)) 1 else 0 };
}
```

```awk
# parse_request_zig(poll_result, req)
# Parses hawk_net_poll() output into req[].
# poll_result format: conn_id RS method RS path RS headers_block RS body_len RS body
# RS = \x1e (ASCII Record Separator, 0x1E)
# Returns 1 on success, 0 on failure.
function parse_request_zig(poll_result, req,    RS_CHAR, parts, n, i, j, k, v, colon, hlines, hn, body, q) {
  RS_CHAR = "\x1e"
  delete req
  n = split(poll_result, parts, RS_CHAR)
  if (n < 5) return 0

  req["_conn_id"]     = parts[1]
  req["method"]       = parts[2]
  req["full_path"]    = parts[3]
  req["http_version"] = "HTTP/1.1"

  # Parse headers block (parts[4]): "Name: value\r\nName2: value2\r\n"
  hn = split(parts[4], hlines, "\r\n")
  for (j = 1; j <= hn; j++) {
    if (hlines[j] == "") continue
    colon = index(hlines[j], ":")
    if (colon == 0) continue
    k = to_lower(trim(substr(hlines[j], 1, colon - 1)))
    v = trim(substr(hlines[j], colon + 1))
    req["header:" k] = v
  }

  # Reconstruct body: parts[6..n] joined with RS_CHAR
  # (parts[5] = body_len, used for validation only)
  body = ""
  for (i = 6; i <= n; i++) {
    body = body (i > 6 ? RS_CHAR : "") parts[i]
  }
  req["body"] = body
  req["raw"]  = poll_result

  # Path / query split
  q = index(req["full_path"], "?")
  if (q > 0) {
    req["path"]         = substr(req["full_path"], 1, q - 1)
    req["query_string"] = substr(req["full_path"], q + 1)
    _request_parse_kv(req["query_string"], req, "query:")
  } else {
    req["path"]         = req["full_path"]
    req["query_string"] = ""
  }

  # Body parsing
  if (length(req["body"]) > 0) {
    if (index(req["header:content-type"], "application/x-www-form-urlencoded") == 1)
      _request_parse_kv(req["body"], req, "form:")
    else if (index(req["header:content-type"], "application/json") == 1)
      _request_parse_json_body(req["body"], req)
  }

  if (req["method"] !~ /^[A-Z]+$/) return 0
  if (req["path"] == "") return 0
  return 1
}
```

```awk
    hdr_size = 0
    while (1) {
      ok = (sock |& getline line)
      if (ok <= 0) break
      sub(/\r$/, "", line)
      if (line == "") break
      headers_raw = headers_raw line "\r\n"
      hdr_size += length(line) + 2
      if (hdr_size > HAWK_MAX_HEADER_SIZE) {
        http_send_simple(sock, 431, "Request Header Fields Too Large")
        close(sock)
        headers_raw = ""
        break
      }
    }

    if (headers_raw == "") {
      close(sock)
      continue
    }

    start_ms = now_ms()
    body = ""
    content_length = http_header_get(headers_raw, "Content-Length") + 0
    if (content_length > HAWK_MAX_BODY_SIZE) {
      http_send_simple(sock, 413, "Payload Too Large")
      close(sock)
      continue
    }
    if (content_length > 0) {
      body = http_read_body(sock, content_length)
    }

    raw = headers_raw "\r\n" body
    delete req
    delete res
    delete _ARR_COUNT
    res["status"] = 200
    if (!request_parse(raw, req)) {
      http_send_simple(sock, 400, "Bad Request")
      close(sock)
      continue
    }

    if (call_hooks("pre_request", req, res) == 1) {
      http_send(sock, res, req, start_ms)
      close(sock)
      continue
    }

    dispatch_ok = router_dispatch(req, res)
    if (dispatch_ok == 0) {
      if (!serve_static(req, res)) {
        status(res, 404)
        text(res, "Not Found")
      }
    }

    call_hooks("post_request", req, res)
    http_send(sock, res, req, start_ms)
    close(sock)
  }

  close(sock)
}

# _http_serve_zig() -- Zig async path (libs/net loaded)
function _http_serve_zig(    poll_result, req, res, conn_id, dispatch_ok, start_ms, parts, n) {
  if (!hawk_net_listen(HAWK_PORT)) {
    log_warn(sprintf("libs/net: failed to bind port %d, falling back to /inet/tcp/", HAWK_PORT))
    _http_serve_inet()
    return
  }

  while (!HAWK_SHUTDOWN) {
    poll_result = hawk_net_poll()
    if (poll_result == "") {
      log_error("libs/net: event loop stopped unexpectedly")
      break
    }

    start_ms = now_ms()
    delete req
    delete res
    delete _ARR_COUNT
    res["status"] = 200

    if (!parse_request_zig(poll_result, req)) {
      n = split(poll_result, parts, "\x1e")
      conn_id = (n >= 1) ? parts[1] : "0"
      _zig_http_send_simple(conn_id, 400, "Bad Request")
      continue
    }

    conn_id = req["_conn_id"]

    if (call_hooks("pre_request", req, res) == 1) {
      _zig_http_send(conn_id, res, req, start_ms)
      continue
    }

    dispatch_ok = router_dispatch(req, res)
    if (dispatch_ok == 0) {
      if (!serve_static(req, res)) {
        status(res, 404)
        text(res, "Not Found")
      }
    }

    call_hooks("post_request", req, res)
    _zig_http_send(conn_id, res, req, start_ms)
  }
}

# _zig_http_send(conn_id, res, req, start_ms)
# Mirrors http_send() but calls hawk_net_respond() instead of writing to /inet/tcp/.
function _zig_http_send(conn_id, res, req, start_ms,    wire, split_pos, head_part, body_part, first_crlf, status_line, headers, content, dur, ts) {
  if (res["sent"]) return

  if (res["_binary_path"] != "" && LIBS_LOADED["binary"]) {
    content = hawk_bin_read(res["_binary_path"])
    res["body"] = ""
    res["header:content-length"] = hawk_bin_length(content)
    wire = response_wire(res)
    split_pos = index(wire, "\r\n\r\n")
    head_part = substr(wire, 1, split_pos - 1)
    first_crlf = index(head_part, "\r\n")
    status_line = substr(head_part, 1, first_crlf - 1)
    headers = substr(head_part, first_crlf + 2) "\r\n"
    hawk_net_respond(conn_id, status_line, headers, content)
  } else {
    wire = response_wire(res)
    split_pos = index(wire, "\r\n\r\n")
    head_part = substr(wire, 1, split_pos - 1)
    body_part = substr(wire, split_pos + 4)
    first_crlf = index(head_part, "\r\n")
    status_line = substr(head_part, 1, first_crlf - 1)
    headers = substr(head_part, first_crlf + 2) "\r\n"
    hawk_net_respond(conn_id, status_line, headers, body_part)
  }
  res["sent"] = 1

  dur = now_ms() - start_ms
  ts  = strftime("%Y-%m-%dT%H:%M:%S%z")
  printf "%s\tINFO\t%s\t%s\t%d\t%d\n", ts, req["method"], req["path"], res["status"], dur
  fflush()
}

# _zig_http_send_simple(conn_id, code, body)
# Mirrors http_send_simple() for the Zig path.
function _zig_http_send_simple(conn_id, code, body) {
  hawk_net_respond(conn_id,
    "HTTP/1.1 " code " " response_status_text(code),
    "Content-Type: text/plain; charset=utf-8\r\nContent-Length: " length(body) "\r\n",
    body)
}
```
