**Tech Stack:** Zig 0.16, `std.crypto` (sha2, hmac, pwhash.argon2, sign.ecdsa), `gawkapi.h`, gawk 5.x extension API.

```zig
// SPDX-License-Identifier: MIT
// libs/multipart/src/multipart.zig
// Pure-Zig multipart/form-data parser. No gawk dependency.
const std = @import("std");

pub const Part = struct {
    name: []u8,
    filename: []u8,
    content_type: []u8,
    body: []u8,
};

pub fn freeParts(allocator: std.mem.Allocator, parts: []Part) void {
    for (parts) |p| {
        allocator.free(p.name);
        allocator.free(p.filename);
        allocator.free(p.content_type);
        allocator.free(p.body);
    }
    allocator.free(parts);
}

pub fn parse(allocator: std.mem.Allocator, body: []const u8, boundary: []const u8) ![]Part {
    // Delimiter: "--" + boundary + "\r\n" opens first part
    const open = try std.mem.concat(allocator, u8, &.{ "--", boundary, "\r\n" });
    defer allocator.free(open);
    // Between parts: "\r\n--" + boundary + "\r\n"
    const mid = try std.mem.concat(allocator, u8, &.{ "\r\n--", boundary });
    defer allocator.free(mid);

    const first = std.mem.indexOf(u8, body, open) orelse return error.NoBoundary;
    var remaining = body[first + open.len ..];

    var list = std.ArrayList(Part).init(allocator);
    errdefer {
        for (list.items) |p| {
            allocator.free(p.name);
            allocator.free(p.filename);
            allocator.free(p.content_type);
            allocator.free(p.body);
        }
        list.deinit();
    }

    while (remaining.len > 0) {
        const end = std.mem.indexOf(u8, remaining, mid) orelse {
            // Last part (no mid delimiter found)
            if (remaining.len > 0) {
                const p = try parsePart(allocator, remaining);
                try list.append(p);
            }
            break;
        };
        const p = try parsePart(allocator, remaining[0..end]);
        try list.append(p);
        remaining = remaining[end + mid.len ..];
        // After mid: either "--\r\n" (next part) or "--" (final boundary)
        if (std.mem.startsWith(u8, remaining, "--")) break;
        if (std.mem.startsWith(u8, remaining, "\r\n")) remaining = remaining[2..];
    }

    return list.toOwnedSlice(allocator);
}

fn parsePart(allocator: std.mem.Allocator, part: []const u8) !Part {
    const sep = "\r\n\r\n";
    const header_end = std.mem.indexOf(u8, part, sep) orelse return error.MalformedPart;
    const headers_raw = part[0..header_end];
    const body = part[header_end + sep.len ..];

    var name: []const u8 = "";
    var filename: []const u8 = "";
    var content_type: []const u8 = "application/octet-stream";

    var line_iter = std.mem.splitSequence(u8, headers_raw, "\r\n");
    while (line_iter.next()) |line| {
        const colon = std.mem.indexOfScalar(u8, line, ':') orelse continue;
        const hname = std.mem.trim(u8, line[0..colon], " ");
        const hval = std.mem.trim(u8, line[colon + 1 ..], " ");

        if (std.ascii.eqlIgnoreCase(hname, "content-disposition")) {
            name = extractParam(hval, "name") orelse "";
            filename = extractParam(hval, "filename") orelse "";
        } else if (std.ascii.eqlIgnoreCase(hname, "content-type")) {
            content_type = hval;
        }
    }

    return Part{
        .name = try allocator.dupe(u8, name),
        .filename = try allocator.dupe(u8, filename),
        .content_type = try allocator.dupe(u8, content_type),
        .body = try allocator.dupe(u8, body),
    };
}

/// Extract param value from Content-Disposition header value.
/// e.g. extractParam(`form-data; name="foo"`, "name") → "foo"
fn extractParam(header: []const u8, param: []const u8) ?[]const u8 {
    // Search for `param="` pattern
    var search_buf: [64]u8 = undefined;
    const needle = std.fmt.bufPrint(&search_buf, "{s}=\"", .{param}) catch return null;
    const pos = std.mem.indexOf(u8, header, needle) orelse return null;
    const value_start = pos + needle.len;
    const value_end = std.mem.indexOfScalarPos(u8, header, value_start, '"') orelse return null;
    return header[value_start..value_end];
}
```

```zig
// SPDX-License-Identifier: MIT
// libs/multipart/src/root.zig
const std = @import("std");
const ffi = @import("gawk_ffi");
const multipart = @import("multipart");

const _ffi_entry = ffi.makeDlLoad(.{
    .name = "hawk_multipart",
    .functions = &.{
        .{ .name = "hawk_multipart_parse", .impl = &multipartParse, .args = 3 },
    },
});
comptime {
    _ = _ffi_entry;
}

// hawk_multipart_parse(body, boundary, req_array) → "1" (ok) / "0" (error)
fn multipartParse(args: ffi.Args) ffi.Result {
    const body = args.getString(0);
    const boundary = args.getString(1);
    const arr = args.getArray(2);
    if (arr == null) return .{ .string = "0" };

    const alloc = std.heap.c_allocator;
    const parts = multipart.parse(alloc, body, boundary) catch return .{ .string = "0" };
    defer multipart.freeParts(alloc, parts);

    for (parts) |part| {
        if (part.name.len == 0) continue;

        if (part.filename.len > 0) {
            // File field: write content, filename, content-type
            const file_key = std.fmt.allocPrint(alloc, "file:{s}", .{part.name}) catch continue;
            defer alloc.free(file_key);
            ffi.arraySet(arr, file_key, part.body);

            const fn_key = std.fmt.allocPrint(alloc, "file:{s}_filename", .{part.name}) catch continue;
            defer alloc.free(fn_key);
            ffi.arraySet(arr, fn_key, part.filename);

            const ct_key = std.fmt.allocPrint(alloc, "file:{s}_type", .{part.name}) catch continue;
            defer alloc.free(ct_key);
            ffi.arraySet(arr, ct_key, part.content_type);
        } else {
            // Text field
            const form_key = std.fmt.allocPrint(alloc, "form:{s}", .{part.name}) catch continue;
            defer alloc.free(form_key);
            ffi.arraySet(arr, form_key, part.body);
        }
    }

    return .{ .string = "1" };
}
```

```awk
# Extract boundary value from Content-Type header.
# Input:  "multipart/form-data; boundary=----WebKitFormBoundaryXYZ"
# Output: "----WebKitFormBoundaryXYZ"
function _extract_boundary(ct,    i, part, parts, n) {
  n = split(ct, parts, ";")
  for (i = 1; i <= n; i++) {
    part = parts[i]
    if (index(part, "boundary=") > 0) {
      sub(/.*boundary=/, "", part)
      gsub(/^[ \t]+|[ \t]+$/, "", part)
      return part
    }
  }
  return ""
}
```

```zig
// SPDX-License-Identifier: MIT
// libs/crypto/src/crypto.zig
// Pure-Zig crypto primitives. No gawk dependency.
const std = @import("std");

const Sha256 = std.crypto.hash.sha2.Sha256;
const HmacSha256 = std.crypto.auth.hmac.sha2.HmacSha256;
const Argon2 = std.crypto.pwhash.argon2;
const Ecdsa = std.crypto.sign.ecdsa.EcdsaP256Sha256;
const base64url = std.base64.url_safe_no_pad;

// sha256(data) → lowercase hex string [64]u8
pub fn sha256(data: []const u8) [64]u8 {
    var digest: [Sha256.digest_length]u8 = undefined;
    Sha256.hash(data, &digest, .{});
    return bytesToHex(&digest);
}

// hmacSha256(key, data) → lowercase hex string [64]u8
pub fn hmacSha256(key: []const u8, data: []const u8) [64]u8 {
    var mac: [HmacSha256.mac_length]u8 = undefined;
    HmacSha256.create(&mac, data, key);
    return bytesToHex(&mac);
}

// argon2idHash(allocator, password) → allocated "$argon2id$..." string
// salt is generated internally by std.crypto.pwhash.argon2.strHash
pub fn argon2idHash(allocator: std.mem.Allocator, password: []const u8) ![]u8 {
    var buf: [256]u8 = undefined;
    const encoded = try Argon2.strHash(password, .{
        .allocator = allocator,
        .params = .{ .t = 2, .m = 65536, .p = 1 },
    }, &buf);
    return allocator.dupe(u8, encoded);
}

// argon2idVerify(allocator, encoded_hash, password) → bool
pub fn argon2idVerify(allocator: std.mem.Allocator, hash: []const u8, password: []const u8) bool {
    Argon2.strVerify(hash, password, .{ .allocator = allocator }) catch return false;
    return true;
}

// Key pair for testing: raw bytes for secret and public key
pub const EcdsaKeyPair = struct {
    secret_key_bytes: [Ecdsa.SecretKey.encoded_length]u8,
    public_key_bytes: [Ecdsa.PublicKey.uncompressed_sec1_encoded_length]u8,
};

pub fn ecdsaGenKey() !EcdsaKeyPair {
    const kp = try Ecdsa.KeyPair.create(null);
    return .{
        .secret_key_bytes = kp.secret_key.toBytes(),
        .public_key_bytes = kp.public_key.toUncompressedSec1(),
    };
}

// ecdsaSign(allocator, secret_key_bytes[32], data) → base64url-encoded sig (alloc'd)
pub fn ecdsaSign(allocator: std.mem.Allocator, secret_key_bytes: []const u8, data: []const u8) ![]u8 {
    if (secret_key_bytes.len != Ecdsa.SecretKey.encoded_length) return error.InvalidKeyLength;
    const sk = try Ecdsa.SecretKey.fromBytes(secret_key_bytes[0..Ecdsa.SecretKey.encoded_length].*);
    const kp = try Ecdsa.KeyPair.fromSecretKey(sk);
    const sig = try kp.sign(data, null);
    const sig_bytes = sig.toBytes();
    const enc_len = base64url.Encoder.calcSize(sig_bytes.len);
    const buf = try allocator.alloc(u8, enc_len);
    _ = base64url.Encoder.encode(buf, &sig_bytes);
    return buf;
}

// ecdsaVerify(allocator, public_key_bytes[65], data, base64url_sig) → bool
pub fn ecdsaVerify(allocator: std.mem.Allocator, pub_key_bytes: []const u8, data: []const u8, b64_sig: []const u8) bool {
    _ = allocator;
    if (pub_key_bytes.len != Ecdsa.PublicKey.uncompressed_sec1_encoded_length) return false;
    const pk = Ecdsa.PublicKey.fromSec1(pub_key_bytes) catch return false;
    const sig_len = base64url.Decoder.calcSizeForSlice(b64_sig) catch return false;
    if (sig_len != Ecdsa.Signature.encoded_length) return false;
    var sig_bytes: [Ecdsa.Signature.encoded_length]u8 = undefined;
    base64url.Decoder.decode(&sig_bytes, b64_sig) catch return false;
    const sig = Ecdsa.Signature.fromBytes(sig_bytes);
    sig.verify(data, pk) catch return false;
    return true;
}

// Internal: bytes → lowercase hex string
fn bytesToHex(bytes: []const u8) [64]u8 {
    const hex_chars = "0123456789abcdef";
    var out: [64]u8 = undefined;
    for (bytes, 0..) |b, i| {
        out[i * 2] = hex_chars[b >> 4];
        out[i * 2 + 1] = hex_chars[b & 0xf];
    }
    return out;
}
```

> **Note on `bytesToHex`:** SHA-256 produces 32 bytes → 64 hex chars. HMAC-SHA256 also produces 32 bytes → 64 hex chars. The function returns `[64]u8` which is correct for both.

```zig
const _alloc = std.heap.c_allocator;

const _ffi_entry = ffi.makeDlLoad(.{

        .{ .name = "hawk_hmac_sha256",      .impl = &hmacSha256Fn,     .args = 2 },
        .{ .name = "hawk_argon2id",         .impl = &argon2idFn,       .args = 1 },
        .{ .name = "hawk_argon2id_verify",  .impl = &argon2idVerifyFn, .args = 2 },
        .{ .name = "hawk_ecdsa_sign",       .impl = &ecdsaSignFn,      .args = 2 },
        .{ .name = "hawk_ecdsa_verify",     .impl = &ecdsaVerifyFn,    .args = 3 },
    },
});
comptime {
    _ = _ffi_entry;
}

// hawk_sha256(data) → hex string (64 chars)
fn sha256Fn(args: ffi.Args) ffi.Result {
    const data = args.getString(0);
    const hex = crypto.sha256(data);
    const out = ffi.gawkAllocator().dupe(u8, &hex) catch return .none;
    return .{ .string = out };
}

// hawk_hmac_sha256(key, data) → hex string (64 chars)
fn hmacSha256Fn(args: ffi.Args) ffi.Result {
    const key = args.getString(0);
    const data = args.getString(1);
    const hex = crypto.hmacSha256(key, data);
    const out = ffi.gawkAllocator().dupe(u8, &hex) catch return .none;
    return .{ .string = out };
}

// hawk_argon2id(password) → "$argon2id$..." encoded string
fn argon2idFn(args: ffi.Args) ffi.Result {
    const password = args.getString(0);
    const hash = crypto.argon2idHash(_alloc, password) catch return .none;
    defer _alloc.free(hash);
    const out = ffi.gawkAllocator().dupe(u8, hash) catch return .none;
    return .{ .string = out };
}

// hawk_argon2id_verify(encoded_hash, password) → "1" or "0"
fn argon2idVerifyFn(args: ffi.Args) ffi.Result {
    const hash = args.getString(0);
    const password = args.getString(1);
    return .{ .string = if (crypto.argon2idVerify(_alloc, hash, password)) "1" else "0" };
}

// hawk_ecdsa_sign(secret_key_b64url, data) → base64url signature
// secret_key_b64url: base64url-encoded 32-byte raw P-256 secret key
fn ecdsaSignFn(args: ffi.Args) ffi.Result {
    const sk_b64 = args.getString(0);
    const data = args.getString(1);
    const b64 = std.base64.url_safe_no_pad;
    const sk_len = b64.Decoder.calcSizeForSlice(sk_b64) catch return .none;
    if (sk_len != 32) return .none;
    var sk_bytes: [32]u8 = undefined;
    b64.Decoder.decode(&sk_bytes, sk_b64) catch return .none;
    const sig = crypto.ecdsaSign(_alloc, &sk_bytes, data) catch return .none;
    defer _alloc.free(sig);
    const out = ffi.gawkAllocator().dupe(u8, sig) catch return .none;
    return .{ .string = out };
}

// hawk_ecdsa_verify(pub_key_b64url, data, sig_b64url) → "1" or "0"
// pub_key_b64url: base64url-encoded 65-byte uncompressed SEC1 public key
fn ecdsaVerifyFn(args: ffi.Args) ffi.Result {
    const pk_b64 = args.getString(0);
    const data = args.getString(1);
    const sig_b64 = args.getString(2);
    const b64 = std.base64.url_safe_no_pad;
    const pk_len = b64.Decoder.calcSizeForSlice(pk_b64) catch return .{ .string = "0" };
    if (pk_len != 65) return .{ .string = "0" };
    var pk_bytes: [65]u8 = undefined;
    b64.Decoder.decode(&pk_bytes, pk_b64) catch return .{ .string = "0" };
    const ok = crypto.ecdsaVerify(_alloc, &pk_bytes, data, sig_b64);
    return .{ .string = if (ok) "1" else "0" };
}
```
