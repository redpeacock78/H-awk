# hawk DSL レイヤー 実装プラン

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** hawk DSL プリプロセッサを実装する — `hawk.app.get(...)` のようなドット記法と `let` ローカル変数宣言を純粋な gawk コードに変換する。

**Architecture:** gawk 製ライン指向ステートマシン（`dsl/desugar.awk`）が `@include` で分割されたモジュールを束ね、入力ファイルを1パスで処理して純粋な gawk スクリプトを stdout に出力する。`bin/hawk` は mktemp で変換後コードを一時ファイルに書き出して実行する。

**Tech Stack:** gawk 4.2+, bash 4.3+（既存要件）, GNU diff（テスト）

```awk
# SPDX-License-Identifier: MIT
# dsl/desugar_dot.awk -- dot-notation → ns::dispatch(path, args) transform

function _ds_dot_transform(line,    result, segs, n, i) {
  n = _ds_split_code_segs(line, segs)
  result = ""
  for (i = 1; i <= n; i++) {
    if (segs[i, "safe"])
      result = result _ds_dot_transform_code(segs[i, "text"])
    else
      result = result segs[i, "text"]
  }
  return result
}

function _ds_dot_transform_code(seg,    result, m, ns, path, args, n, parts, i) {
  result = ""
  while (match(seg, /[a-zA-Z_][a-zA-Z0-9_]*(\.[a-zA-Z_][a-zA-Z0-9_]*)+\(/)) {
    result = result substr(seg, 1, RSTART - 1)
    m   = substr(seg, RSTART, RLENGTH - 1)   # matched chain without "("
    seg = substr(seg, RSTART + RLENGTH)       # rest after "("
    n   = split(m, parts, ".")
    ns  = parts[1]
    path = ""
    for (i = 2; i <= n; i++)
      path = path (i > 2 ? "." : "") parts[i]
    args = _ds_extract_args(seg)
    seg  = substr(seg, length(args) + 2)      # skip args + ")"
    result = result ns "::dispatch(\"" path "\"" (args != "" ? ", " args : "") ")"
  }
  return result seg
}

function _ds_extract_args(str,    depth, i, c, n, chars, args) {
  depth = 1; args = ""
  n = split(str, chars, "")
  for (i = 1; i <= n; i++) {
    c = chars[i]
    if      (c == "(") depth++
    else if (c == ")") { depth--; if (depth == 0) break }
    args = args c
  }
  return args
}
```

```awk
# SPDX-License-Identifier: MIT
# dsl/desugar.awk -- hawk DSL preprocessor entry point

@include "dsl/desugar_state.awk"
@include "dsl/desugar_strings.awk"
@include "dsl/desugar_dot.awk"
@include "dsl/desugar_let.awk"

BEGIN { _ds_init() }

FNR == 1 {
  _DS_src_file = FILENAME
  print "# line 1 \"" FILENAME "\""
}

{ _ds_process_line($0, FNR) }

END {
  if (_DS_had_error) exit 1
  if (_DS_in_function) {
    print "dsl error: " _DS_src_file ": unclosed function '" _DS_func_name "'" \
      > "/dev/stderr"
    exit 1
  }
}

function _ds_process_line(line, lineno,    transformed) {
  if (!_DS_in_function) {
    if (line ~ /^[[:space:]]*let[[:space:]]/) {
      print "dsl error: " _DS_src_file ":" lineno \
        ": 'let' outside function body" > "/dev/stderr"
      _DS_had_error = 1
      exit 1
    }
    if (_ds_is_func_def(line)) {
      _DS_in_function  = 1
      _DS_func_sig     = line
      _DS_brace_depth  = _ds_net_braces(line)
      _DS_let_count    = 0
      _DS_body_count   = 0
      delete _DS_let_locals
      delete _DS_body_buf
      return
    }
    print _ds_dot_transform(line)
    return
  }

  # Inside function body
  _DS_brace_depth += _ds_net_braces(line)

  if (_DS_brace_depth <= 0) {
    _DS_in_function = 0
    print _ds_rewrite_sig(_DS_func_sig)
    for (i = 1; i <= _DS_body_count; i++) print _DS_body_buf[i]
    print _ds_dot_transform(line)
    return
  }

  transformed = _ds_let_transform(_ds_dot_transform(line), lineno)
  if (transformed != "") _DS_body_buf[++_DS_body_count] = transformed
}

function _ds_is_func_def(line) {
  return (line ~ /^[[:space:]]*function[[:space:]]+[a-zA-Z_][a-zA-Z0-9_]*[[:space:]]*\([^)]*\)[[:space:]]*\{/)
}

function _ds_net_braces(line,    n, i, c) {
  n = 0
  for (i = 1; i <= length(line); i++) {
    c = substr(line, i, 1)
    if      (c == "{") n++
    else if (c == "}") n--
  }
  return n
}
```

```awk
# SPDX-License-Identifier: MIT
# dsl/desugar_let.awk -- let declaration transform + function signature hoisting

function _ds_let_transform(line, lineno,    m) {
  # let name = []  →  delete name
  if (match(line, /^([[:space:]]*)let[[:space:]]+([a-zA-Z_][a-zA-Z0-9_]*)[[:space:]]*=[[:space:]]*\[\][[:space:]]*$/, m)) {
    _DS_let_locals[++_DS_let_count] = m[2]
    return m[1] "delete " m[2]
  }
  # let name = expr  →  name = expr
  if (match(line, /^([[:space:]]*)let[[:space:]]+([a-zA-Z_][a-zA-Z0-9_]*)[[:space:]]*=/, m)) {
    _DS_let_locals[++_DS_let_count] = m[2]
    sub(/let[[:space:]]+/, "", line)
    return line
  }
  # let name  (bare)  →  delete line
  if (match(line, /^([[:space:]]*)let[[:space:]]+([a-zA-Z_][a-zA-Z0-9_]*)[[:space:]]*$/, m)) {
    _DS_let_locals[++_DS_let_count] = m[2]
    return ""
  }
  return line
}

function _ds_rewrite_sig(sig,    locals, i) {
  if (_DS_let_count == 0) return sig
  locals = ""
  for (i = 1; i <= _DS_let_count; i++)
    locals = locals (i > 1 ? ", " : "") _DS_let_locals[i]
  if (sig ~ /\([[:space:]]*\)/) {
    sub(/\([[:space:]]*\)/, "(    " locals ")", sig)
  } else {
    sub(/\)[[:space:]]*\{[[:space:]]*$/, ",    " locals ") {", sig)
  }
  return sig
}
```

```awk
# SPDX-License-Identifier: MIT
# dsl/desugar_strings.awk -- string/comment region detection

# Splits line into segments with safe=1 (code) or safe=0 (string/comment).
# Handles double-quoted strings and # comments.
# Escape sequences (backslash) inside strings are handled.
function _ds_split_code_segs(line, segs,    n, i, c, prev_c, in_str, seg_start) {
  delete segs
  n         = 0
  in_str    = 0
  seg_start = 1
  prev_c    = ""

  for (i = 1; i <= length(line); i++) {
    c = substr(line, i, 1)

    if (!in_str) {
      if (c == "\"") {
        # flush code segment before string
        if (i > seg_start) {
          segs[++n, "safe"] = 1
          segs[n,   "text"] = substr(line, seg_start, i - seg_start)
        }
        in_str    = 1
        seg_start = i
      } else if (c == "#") {
        # flush code segment, rest is comment (safe=0)
        if (i > seg_start) {
          segs[++n, "safe"] = 1
          segs[n,   "text"] = substr(line, seg_start, i - seg_start)
        }
        segs[++n, "safe"] = 0
        segs[n,   "text"] = substr(line, i)
        return n
      }
    } else {
      # inside string: end on unescaped "
      if (c == "\"" && prev_c != "\\") {
        segs[++n, "safe"] = 0
        segs[n,   "text"] = substr(line, seg_start, i - seg_start + 1)
        in_str    = 0
        seg_start = i + 1
      }
    }
    prev_c = (c == "\\" && prev_c == "\\") ? "" : c
  }

  # flush remaining
  if (seg_start <= length(line)) {
    segs[++n, "safe"] = (in_str ? 0 : 1)
    segs[n,   "text"] = substr(line, seg_start)
  }
  return n
}
```

```bash
# Desugar DSL → pure gawk via mktemp
HAWK_DSL_DIR="$(dirname "$0")/../dsl"
DESUGAR_SCRIPT="${HAWK_DSL_DIR}/desugar.awk"
_hawk_desugar() {
  local src="$1"
  local tmp
  tmp=$(mktemp /tmp/hawk.XXXXXX.awk)
  if [[ -z "${HAWK_DEBUG}" ]]; then
    trap "rm -f '$tmp'" EXIT
  fi
  if ! gawk -f "${DESUGAR_SCRIPT}" "$src" > "$tmp"; then
    echo "[hawk] desugar failed" >&2
    exit 1
  fi
  if [[ ! -s "$tmp" ]]; then
    echo "[hawk] desugar produced empty output" >&2
    exit 1
  fi
  printf '%s' "$tmp"
}

if [[ -f "${DESUGAR_SCRIPT}" ]]; then
  APP=$(_hawk_desugar "$APP")
  _HAWK_TMP_APP="$APP"
fi
```

- [ ] **Step 1: DSL 記法版 todo アプリ作成**

```awk
# SPDX-License-Identifier: MIT
# examples/dsl/app.awk -- H-awk todo demo in DSL notation
# NOTE: requires hawk::dispatch / ctx::dispatch runtime infrastructure (future work)

BEGIN {
  # let は function body 外では使用不可 — BEGIN 内は通常 AWK 記法を使う
  hawk.app.get("/",           "todo_index")
  hawk.app.get("/todos",      "todo_list_html")
  hawk.app.post("/todos",     "todo_add")
  hawk.app.del("/todos/:id",  "todo_delete")
  hawk.app.listen(hawk.env.has("PORT") ? hawk.env.get("PORT") + 0 : 8080)
}

function todo_index() {
  ctx.res.render("views/index.html")
}

function todo_list_html() {
  let rows = []
  let n = read_tsv("data/todos.tsv", rows)
  let out = ""
  let i
  for (i = 1; i <= n; i++)
    out = out _todo_tr(rows[i, "id"], rows[i, "title"])
  ctx.res.html(out)
}

function todo_add() {
  let title = ctx.req.form("title")
  if (title == "") {
    ctx.res.status(400)
    ctx.res.text("title required")
    return
  }
  let row = []
  row["id"]    = systime() "_" int(rand() * 100000)
  row["title"] = title
  append_tsv("data/todos.tsv", row)
  ctx.res.status(201)
  ctx.res.html(_todo_tr(row["id"], row["title"]))
}

function todo_delete() {
  let deleted = delete_tsv("data/todos.tsv", "id", ctx.req.param("id"))
  if (deleted == 0) {
    ctx.res.status(404)
    ctx.res.text("not found")
    return
  }
  ctx.res.status(200)
  ctx.res.html("")
}

function _todo_tr(id, title) {
  return sprintf("<tr><td>%s</td></tr>", title)
}
```

## 注意事項・既知の制限

- **複数行関数呼び出し**: `_ds_is_func_def` は `function foo() {` が1行に収まる場合のみ検出。`{` が次行にある場合は未対応（初期実装のスコープ外）。
- **brace depth の精度**: `_ds_net_braces` は文字列/コメント内の `{` `}` もカウントする。`desugar_strings.awk` の改良で後から精度向上可能。
- **ネスト関数**: gawk はネスト関数をサポートしないため考慮不要。
- **`@namespace` ディレクティブ**: DSL ユーザーは namespace を意識しない（フレームワーク内部のみ使用）のでスコープ外。
